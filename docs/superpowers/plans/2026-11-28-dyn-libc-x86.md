**Interfaces:**

- Produces: `$(LIBC_BUILD_DIR)/lib/libc.so`(SONAME `/lib/libc.so`);`$(SYSROOT)/usr/lib/libc.so`;静态路径对象/清单零变化

- [ ] **Step 1: `.pic.o` 编译 pass(libc/Makefile,保留既有规则的目录创建与依赖生成)**

```make
# C 源走独立 -fPIC pass;hosted 汇编对象(setjmp/sigreturn_trampoline,
# ARCH_HOSTEDOBJS)原样纳入——评审 P1-2 宿主实测:仅链 sigaction.c 即产
# 指向未定义 sigreturn_trampoline 的 GLOB_DAT,eager -z now 下 ld.so 必须
# 解析全部强符号,缺对象 = 每次动态 exec 必死。
LIBC_PIC_OBJS := $(addprefix $(BUILD_DIR)/, $(LIBC_C_SOURCES:.c=.pic.o)) \
                 $(LIBC_HOSTED_OBJS)
# PIC 编译键独立于 LIBC_COMPILE_KEY(键内容=实际 PIC flags)——否则头文件
# 改动只重编静态 pass、发布旧 libc.so(评审 P2)。沿用 LIBC_KEY_FORCE 的
# FORCE 模式。
LIBC_PIC_COMPILE_KEY := $(BUILD_DIR)/.libc_pic_compile.key
$(LIBC_PIC_COMPILE_KEY): LIBC_KEY_FORCE
	@mkdir -p $(dir $@)
	@printf '%s\n' 'compiler=$(EFFECTIVE_CC) cflags=$(strip $(PIC_PASS_CFLAGS))' > $@.new
	@cmp -s $@.new $@ || mv -f $@.new $@

$(BUILD_DIR)/%.pic.o: %.c $(LIBC_PIC_COMPILE_KEY)
	@mkdir -p $(dir $@)
	$(EFFECTIVE_CC) $(PIC_PASS_CFLAGS) -MMD -MP -c $< -o $@
-include $(LIBC_PIC_OBJS:.o=.d)
```

`PIC_PASS_CFLAGS` = 现有 `%.o` 规则 CFLAGS 原文,**只把 PIE flag 换成 `-fPIC`**(宿主实验核实现规则实际拼写后对齐;`-MMD -MP`/mkdir 与既有对象规则逐字对齐);汇编对象沿用既有 hosted 规则与 ARCH_CFLAGS。**crt0.o 照旧单独发布,不进 .so**。

- [ ] **Step 2: libc.so 链接目标(x86_64 guard 内,带链接键)**

```make
# aarch64 子 Make(HOSTARCH=aarch64)不得进入 elf_x86_64 规则(评审 P1-1)。
ifeq ($(HOSTARCH),x86_64)
# 链接键(普通前置,仿 user/Makefile USER_LINK_KEY):记录有序对象清单 +
# 链接器 + 实际链接参数——删源缩短清单、改 SONAME/flags 不动对象时间戳
# 时,没有键 Make 会跳过链接并把旧 .so 重新发布进 sysroot(评审 P2-1)。
LIBC_SO_LINK_KEY := $(LIBC_BUILD_DIR)/.libc_so_link.key
$(LIBC_SO_LINK_KEY): LIBC_KEY_FORCE
	@mkdir -p $(dir $@)
	@printf '%s\n' 'linker=$(TARGET_LD) objects=$(strip $(LIBC_PIC_OBJS)) ldflags=$(strip $(LIBC_SO_LDFLAGS))' > $@.new
	@cmp -s $@.new $@ || mv -f $@.new $@

LIBC_SO_LDFLAGS := -m elf_x86_64 -shared --hash-style=sysv -z now -z norelro \
  -z max-page-size=4096 --pack-dyn-relocs=none -z defs
$(LIBC_BUILD_DIR)/lib/libc.so: $(LIBC_PIC_OBJS) $(LIBC_SO_LINK_KEY)
	@mkdir -p $(dir $@)
	@rm -f $@
	$(TARGET_LD) $(LIBC_SO_LDFLAGS) \
	  -soname /lib/libc.so -o $@ $(LIBC_PIC_OBJS)
endif
```

**`-z defs` 必加**:`-shared` 默认容忍未定义符号(变成 undefined export),eager 重定位下这是把链接错误推迟到每次 exec 必死;`-z defs` 让缺对象在链接期响亮失败。若链接报重复符号(不应——Task 1 已删)/未定义符号,停下分析,**不得**给 .so 链接引入 libc.a。

- [ ] **Step 3: 构建期断言脚本(随构建验证步骤跑,预期全部通过)**

```bash
SO=build/x86_64/libc/lib/libc.so
llvm-readelf --dyn-symbols $SO | awk '$7=="UND"'          # UNDEF 集
llvm-readelf -d $SO | grep -E "SONAME|FLAGS"              # SONAME=/lib/libc.so
llvm-readelf -S $SO | grep -E "hash|dynsym|dynstr"        # sysv hash 存在
llvm-readelf -n $SO                                       # 无 TEXTREL 的佐证之一
```

断言:① UNDEF 集**无 WEAK**(若有 → 停,按 spec 分支补 ld.so weak-undef→0);② STRONG UNDEF 全部为 syscall 包装/既有 libc 内互调(逐个过目,须都落在 ld.so 符号解析能力内);③ `DT_FLAGS` 无 TEXTREL/静态 TLS 位;④ `llvm-readelf -l` 核对各 PT_LOAD 内存窗 `[align_down(p_vaddr), align_up(p_vaddr+p_memsz))` **互不重叠**、DT_HASH 落无 PF_W LOAD、RELATIVE 重定位 ≥1(`llvm-readelf -r $SO | grep RELATIVE | wc -l`)。把④的数字记入提交信息(RELA/GLOB_DAT/JUMP_SLOT 计数一并),供 Task 4 的 ld.so 循环容量对照。

(`LIBC_SO_PUBLISH` 为 x86_64 分支才定义的依赖变量;aarch64 分支该变量为空,install-libs 形状不变。manifest 是 `find` 扫描,cp 后自动收录,无需改 sysroot.mk 的清单生成。)sysroot.mk 若对 usr/lib 内容有显式清单/stamp,按同目录 libc.a 的挂法补 libc.so 一项。

Run(串联,**保留失败状态**;**直接读发布后的 sysroot**——评审 P2:`.sysroot.next` 会被原子重命名为 `sysroot`,且 `readlink -f` 对不存在的路径只要父目录存在就返回 0,回退分支永不执行):
