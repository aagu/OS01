- 宿主 oracle 单独链接 glibc 的 `-lm`（`-lm` 放在源文件之后）；测试二进制本身不链接 `-lm`。

对拍同时查出并已修正计划里的三处缺陷：

1. `round.c` / `roundf.c` 算了 `frac` 却没先清尾数位，`mag += step` 加在仍带尾数的位模式上 → 结果既不是整数也不是最近的整数（`round(0x40ded3daf70ef129)` 原样返回）。修法：`mag &= ~(step - 1)` 必须排在 `mag += step` 之前。
2. `sqrt.c` / `sqrtf.c` 用符号位直接判负，会把 `-NaN` 误判成域错误并置 `EDOM`；spec 要求 NaN 输入不动 `errno`。修法：先判「任意符号的 NaN」返回固定 quiet NaN，再判负数（含 `-inf`）置 `EDOM`。
3. `fmodf.c` / `sqrtf.c` 原先只写了「同形」的片段；现已补成完整源码。

```c
/* libc/math/musl_libm.h — private helpers for the vendored musl math subset.
 *
 * This header is NOT public: it lives next to the sources that use it and
 * is never installed into the sysroot (only libc/include/ is published).
 * It replaces musl's src/internal/libm.h, which carries musl-internal
 * machinery (long double unions, fp_arch.h, WANT_ROUNDING) that OS01
 * does not have.
 */
#ifndef OS01_LIBM_MUSL_LIBM_H
#define OS01_LIBM_MUSL_LIBM_H

#include <stdint.h>

/* ── bit casts (musl's asuint64/asdouble/asuint/asfloat) ─────── */
static inline double asdouble(uint64_t u)
{
    union { uint64_t i; double d; } v;
    v.i = u;
    return v.d;
}

static inline uint64_t asuint64(double x)
{
    union { uint64_t i; double d; } v;
    v.d = x;
    return v.i;
}

static inline float asfloat(uint32_t u)
{
    union { uint32_t i; float f; } v;
    v.i = u;
    return v.f;
}

static inline uint32_t asuint(float x)
{
    union { uint32_t i; float f; } v;
    v.f = x;
    return v.i;
}

/* ── branch hints (musl's predict_true/predict_false) ─────────── */
#define predict_true(x)  __builtin_expect(!!(x), 1)
#define predict_false(x) __builtin_expect(x, 0)

/* ── the fixed quiet NaN every OS01 math function returns ───────
 *
 * The pattern is positive on purpose: the x86 hardware default NaN
 * carries a negative sign while the ARM one does not, so a fixed
 * positive pattern is the only choice that is identical on both
 * targets.
 */
#define OS01_QNAN_D 0x7ff8000000000000ULL
#define OS01_QNAN_F 0x7fc00000U

#endif /* OS01_LIBM_MUSL_LIBM_H */
```

`libc/math/floor.c`：

```c
/* libc/math/floor.c — integer bit-manipulation floor.
 *
 * Ported from musl 1.2.5 src/math/floorf.c (the float variant is pure
 * bit manipulation there; the double variant uses the 1/EPS toint trick,
 * which depends on the compiler's excess-precision semantics and would
 * drag long double into the userland on x86_64).
 */
#include "musl_libm.h"

double floor(double x)
{
    union {double f; uint64_t i;} u = {x};
    int e = (int)(u.i >> 52 & 0x7ff) - 0x3ff;
    uint64_t m;

    if (e >= 52)
        return x;
    if (e >= 0) {
        m = 0x000fffffffffffffULL >> e;
        if ((u.i & m) == 0)
            return x;
        if (u.i >> 63)
            u.i += m;
        u.i &= ~m;
    } else {
        if (u.i >> 63 == 0)
            u.i = 0;
        else if (u.i << 1)
            u.f = -1.0;
    }
    return u.f;
}
```

`libc/math/round.c`：

```c
/* libc/math/round.c — integer bit-manipulation round (half away from zero).
 *
 * C99 7.12.9.7: round rounds halfway cases away from zero, which is what
 * the fraction comparison below implements without any FP arithmetic.
 */
#include "musl_libm.h"

double round(double x)
{
    union {double f; uint64_t i;} u = {x};
    uint64_t sign = u.i & (1ULL << 63);
    uint64_t mag  = u.i & ~(1ULL << 63);
    int e = (int)(mag >> 52 & 0x7ff) - 0x3ff;

    if (mag == 0)                       /* +-0 */
        return x;
    if (e >= 52)                        /* |x| >= 2^52: already integral */
        return x;
    if (e < 0) {                        /* |x| < 1 */
        if (mag >= 0x3fe0000000000000ULL)   /* |x| >= 0.5 */
            return sign ? -1.0 : 1.0;
        return sign ? -0.0 : 0.0;
    }
    {
        uint64_t step = 1ULL << (52 - e);      /* one ulp at this exponent */
        uint64_t frac = mag & (step - 1);
        mag &= ~(step - 1);                    /* drop the fraction FIRST */
        if (frac >= (step >> 1))               /* >= half: away from zero */
            mag += step;                       /* may carry into the exponent */
        return asdouble(sign | mag);
    }
}
```

> `mag &= ~(step - 1)` 必须在 `mag += step` **之前**：`frac` 判完之后若不清尾数，`mag += step` 加在仍带尾数的位模式上，结果既不是整数也不是最近的整数（宿主对拍实测：`round(0x40ded3daf70ef129)` 会原样返回）。

`libc/math/roundf.c` 同形，尾数 23 位：

```c
/* libc/math/roundf.c — integer bit-manipulation roundf.  See round.c. */
#include "musl_libm.h"

float roundf(float x)
{
    union {float f; uint32_t i;} u = {x};
    uint32_t sign = u.i & (1U << 31);
    uint32_t mag  = u.i & ~(1U << 31);
    int e = (int)(mag >> 23 & 0xff) - 0x7f;

    if (mag == 0)
        return x;
    if (e >= 23)
        return x;
    if (e < 0) {
        if (mag >= 0x3f000000U)            /* |x| >= 0.5 */
            return sign ? -1.0f : 1.0f;
        return sign ? -0.0f : 0.0f;
    }
    {
        uint32_t step = 1U << (23 - e);
        uint32_t frac = mag & (step - 1);
        mag &= ~(step - 1);                /* drop the fraction FIRST */
        if (frac >= (step >> 1))
            mag += step;
        return asfloat(sign | mag);
    }
}
```

`libc/math/fmod.c`（`fmodf.c` 的完整源码紧跟其后）：

```c
/* libc/math/fmod.c — exact remainder by integer long division.
 *
 * musl 1.2.5 src/math/fmod.c, with three adaptations: the NaN test is a
 * bit test (not y != y, which compiles to a compare instruction), the
 * NaN path returns the fixed quiet NaN instead of (x*y)/(x*y), and the
 * signed-zero path builds the zero from the bit pattern instead of 0*x.
 * The error paths follow the errno contract: a NaN operand is not an
 * error, an infinite dividend and a zero divisor are domain errors.
 */
#include <errno.h>
#include "musl_libm.h"

double fmod(double x, double y)
{
    union {double f; uint64_t i;} ux = {x}, uy = {y};
    int ex = ux.i>>52 & 0x7ff;
    int ey = uy.i>>52 & 0x7ff;
    int sx = ux.i>>63;
    uint64_t i;

    uint64_t uxi = ux.i;

    /* NaN operand: not an error, the fixed quiet NaN goes out. */
    if ((ex == 0x7ff && (uxi & 0x000fffffffffffffULL) != 0) ||
        (ey == 0x7ff && (uy.i & 0x000fffffffffffffULL) != 0))
        return asdouble(OS01_QNAN_D);

    /* Domain errors: a zero divisor or an infinite dividend. */
    if ((uy.i<<1 == 0) || ex == 0x7ff) {
        errno = EDOM;
        return asdouble(OS01_QNAN_D);
    }

    if (uxi<<1 <= uy.i<<1) {
        if (uxi<<1 == uy.i<<1) {
            ux.i &= 1ULL << 63;      /* signed zero, no FP multiply */
            return ux.f;
        }
        return x;
    }

    /* normalize x and y */
    if (!ex) {
        for (i = uxi<<12; i>>63 == 0; ex--, i <<= 1);
        uxi <<= -ex + 1;
    } else {
        uxi &= -1ULL >> 12;
        uxi |= 1ULL << 52;
    }
    if (!ey) {
        for (i = uy.i<<12; i>>63 == 0; ey--, i <<= 1);
        uy.i <<= -ey + 1;
    } else {
        uy.i &= -1ULL >> 12;
        uy.i |= 1ULL << 52;
    }

    /* x mod y */
    for (; ex > ey; ex--) {
        i = uxi - uy.i;
        if (i >> 63 == 0) {
            if (i == 0) {
                ux.i &= 1ULL << 63;
                return ux.f;
            }
            uxi = i;
        }
        uxi <<= 1;
    }
    i = uxi - uy.i;
    if (i >> 63 == 0) {
        if (i == 0) {
            ux.i &= 1ULL << 63;
            return ux.f;
        }
        uxi = i;
    }
    for (; uxi>>52 == 0; uxi <<= 1, ex--);

    /* scale result */
    if (ex > 0) {
        uxi -= 1ULL << 52;
        uxi |= (uint64_t)ex << 52;
    } else {
        uxi >>= -ex + 1;
    }
    uxi |= (uint64_t)sx << 63;
    ux.i = uxi;
    return ux.f;
}
```

`libc/math/fmodf.c`（musl 的 `fmodf.c` 用 23 位尾数、32 位长除，改写点与 `fmod.c` 一一对应）：

```c
/* libc/math/fmodf.c — exact remainder by integer long division.
 *
 * musl 1.2.5 src/math/fmodf.c, with the same three adaptations as fmod.c:
 * the NaN test is a bit test (not y != y, which compiles to a compare
 * instruction), the NaN path returns the fixed quiet NaN instead of
 * (x*y)/(x*y), and the signed-zero path builds the zero from the bit
 * pattern instead of 0*x.  The error paths follow the errno contract:
 * a NaN operand is not an error, an infinite dividend and a zero
 * divisor are domain errors.
 */
#include <errno.h>
#include "musl_libm.h"

float fmodf(float x, float y)
{
    union {float f; uint32_t i;} ux = {x}, uy = {y};
    int ex = ux.i>>23 & 0xff;
    int ey = uy.i>>23 & 0xff;
    uint32_t sx = ux.i & 0x80000000;
    uint32_t i;
    uint32_t uxi = ux.i;

    /* NaN operand: not an error, the fixed quiet NaN goes out. */
    if ((ex == 0xff && (uxi & 0x007fffffU) != 0) ||
        (ey == 0xff && (uy.i & 0x007fffffU) != 0))
        return asfloat(OS01_QNAN_F);

    /* Domain errors: a zero divisor or an infinite dividend. */
    if ((uy.i<<1 == 0) || ex == 0xff) {
        errno = EDOM;
        return asfloat(OS01_QNAN_F);
    }

    if (uxi<<1 <= uy.i<<1) {
        if (uxi<<1 == uy.i<<1) {
            ux.i &= 1U << 31;         /* signed zero, no FP multiply */
            return ux.f;
        }
        return x;
    }

    /* normalize x and y */
    if (!ex) {
        for (i = uxi<<9; i>>31 == 0; ex--, i <<= 1);
        uxi <<= -ex + 1;
    } else {
        uxi &= -1U >> 9;
        uxi |= 1U << 23;
    }
    if (!ey) {
        for (i = uy.i<<9; i>>31 == 0; ey--, i <<= 1);
        uy.i <<= -ey + 1;
    } else {
        uy.i &= -1U >> 9;
        uy.i |= 1U << 23;
    }

    /* x mod y */
    for (; ex > ey; ex--) {
        i = uxi - uy.i;
        if (i >> 31 == 0) {
            if (i == 0) {
                ux.i &= 1U << 31;
                return ux.f;
            }
            uxi = i;
        }
        uxi <<= 1;
    }
    i = uxi - uy.i;
    if (i >> 31 == 0) {
        if (i == 0) {
            ux.i &= 1U << 31;
            return ux.f;
        }
        uxi = i;
    }
    for (; uxi>>23 == 0; uxi <<= 1, ex--);

    /* scale result up */
    if (ex > 0) {
        uxi -= 1U << 23;
        uxi |= (uint32_t)ex << 23;
    } else {
        uxi >>= -ex + 1;
    }
    uxi |= sx;
    ux.i = uxi;
    return ux.f;
}
```

`libc/math/sqrt.c`：

```c
/* libc/math/sqrt.c — correctly rounded square root by fixed-point
 * Goldschmidt iteration (musl 1.2.5 src/math/sqrt.c).
 *
 * Adaptations: FENV_SUPPORT is 0 (OS01 has no fenv, so the final
 * rounding-mode nudge is gone), the invalid path returns the fixed
 * quiet NaN and sets EDOM for a negative argument, and the subnormal
 * normalization rebuilds the bit pattern with integer operations
 * instead of the x * 0x1p52 multiply.
 */
#include <errno.h>
#include "musl_libm.h"
#include "sqrt_data.h"

static inline uint32_t mul32(uint32_t a, uint32_t b)
{
    return (uint64_t)a*b >> 32;
}

static inline uint64_t mul64(uint64_t a, uint64_t b)
{
    uint64_t ahi = a>>32;
    uint64_t alo = a&0xffffffff;
    uint64_t bhi = b>>32;
    uint64_t blo = b&0xffffffff;
    return ahi*bhi + (ahi*blo >> 32) + (alo*bhi >> 32);
}

double sqrt(double x)
{
    uint64_t ix, top, m;

    ix = asuint64(x);
    top = ix >> 52;
    if (predict_false(top - 0x001 >= 0x7ff - 0x001)) {
        /* x < 0x1p-1022 or inf or nan.  */
        if (ix * 2 == 0)
            return x;
        if (ix == 0x7ff0000000000000)
            return x;                                  /* +inf */
        if ((ix & 0x7fffffffffffffffULL) > 0x7ff0000000000000ULL)
            return asdouble(OS01_QNAN_D);               /* NaN input, any sign */
        if (ix >> 63) {                                /* negative: domain error */
            errno = EDOM;
            return asdouble(OS01_QNAN_D);
        }
        /* x is subnormal: normalize it with integer operations.
         * The value is m * 2^-1074; the normalized form is
         * (1 + frac) * 2^(p-1074) with p the highest set bit of m. */
        {
            uint64_t mm = ix & 0x000fffffffffffffULL;
            int p = 63 - __builtin_clzll(mm);
            uint64_t frac = (mm - (1ULL << p)) << (52 - p);
            uint64_t biased = (uint64_t)(p + 1);
            ix = (biased << 52) | frac;
            top = ix >> 52;
            top -= 52;
        }    }

    /* argument reduction: x = 4^e m; with integer e, and m in [1, 4) */
    int even = top & 1;
    m = (ix << 11) | 0x8000000000000000;
    if (even) m >>= 1;
    top = (top + 0x3ff) >> 1;

    static const uint64_t three = 0xc0000000;
    uint64_t r, s, d, u, i;

    i = (ix >> 46) % 128;
    r = (uint32_t)__os01_libm_rsqrt_tab[i] << 16;
    s = mul32(m>>32, r);
    d = mul32(s, r);
    u = three - d;
    r = mul32(r, u) << 1;
    s = mul32(s, u) << 1;
    d = mul32(s, r);
    u = three - d;
    r = mul32(r, u) << 1;
    r = r << 32;
    s = mul64(m, r);
    d = mul64(s, r);
    u = (three<<32) - d;
    s = mul64(s, u);  /* repr: 3.61 */
    s = (s - 2) >> 9; /* repr: 12.52 */

    /* compute the nearest rounded result */
    uint64_t d0, d1;
    double y;
    d0 = (m << 42) - s*s;
    d1 = s - d0;
    s += d1 >> 63;
    s &= 0x000fffffffffffff;
    s |= top << 52;
    y = asdouble(s);
    return y;
}
```

`libc/math/sqrtf.c`（musl 的 `sqrtf.c` 是 32 位定点版，次正规数重建后要按 `ix -= 23U << 23` 补偿指数）：

```c
/* libc/math/sqrtf.c — correctly rounded square root by fixed-point
 * Goldschmidt iteration (musl 1.2.5 src/math/sqrtf.c).
 *
 * Adaptations: FENV_SUPPORT is 0 (OS01 has no fenv, so the final
 * rounding-mode nudge is gone), the invalid path returns the fixed
 * quiet NaN and sets EDOM for a negative argument, and the subnormal
 * normalization rebuilds the bit pattern with integer operations
 * instead of the x * 0x1p23f multiply.
 */
#include <errno.h>
#include "musl_libm.h"
#include "sqrt_data.h"

static inline uint32_t mul32(uint32_t a, uint32_t b)
{
    return (uint64_t)a*b >> 32;
}

/* see sqrt.c for more detailed comments.  */

float sqrtf(float x)
{
    uint32_t ix, m, m1, m0, even, ey;

    ix = asuint(x);
    if (predict_false(ix - 0x00800000 >= 0x7f800000 - 0x00800000)) {
        /* x < 0x1p-126 or inf or nan.  */
        if (ix * 2 == 0)
            return x;
        if (ix == 0x7f800000)
            return x;
        if ((ix & 0x7fffffffU) > 0x7f800000U)
            return asfloat(OS01_QNAN_F);           /* NaN input, any sign */
        if (ix >> 31) {
            errno = EDOM;
            return asfloat(OS01_QNAN_F);           /* negative: domain error */
        }
        /* x is subnormal, normalize it.  */
        {
            uint32_t mm = ix & 0x007fffffU;
            int p = 31 - __builtin_clz(mm);
            uint32_t frac = (mm - (1U << p)) << (23 - p);
            uint32_t biased = (uint32_t)(p + 1);
            ix = (biased << 23) | frac;
            ix -= 23U << 23;
        }
    }

    /* x = 4^e m; with int e and m in [1, 4).  */
    even = ix & 0x00800000;
    m1 = (ix << 8) | 0x80000000;
    m0 = (ix << 7) & 0x7fffffff;
    m = even ? m0 : m1;

    /* 2^e is the exponent part of the return value.  */
    ey = ix >> 1;
    ey += 0x3f800000 >> 1;
    ey &= 0x7f800000;

    /* compute r ~ 1/sqrt(m), s ~ sqrt(m) with 2 goldschmidt iterations.  */
    static const uint32_t three = 0xc0000000;
    uint32_t r, s, d, u, i;
    i = (ix >> 17) % 128;
    r = (uint32_t)__os01_libm_rsqrt_tab[i] << 16;
    /* |r*sqrt(m) - 1| < 0x1p-8 */
    s = mul32(m, r);
    /* |s/sqrt(m) - 1| < 0x1p-8 */
    d = mul32(s, r);
    u = three - d;
    r = mul32(r, u) << 1;
    /* |r*sqrt(m) - 1| < 0x1.7bp-16 */
    s = mul32(s, u) << 1;
    /* |s/sqrt(m) - 1| < 0x1.7bp-16 */
    d = mul32(s, r);
    u = three - d;
    s = mul32(s, u);
    /* -0x1.03p-28 < s/sqrt(m) - 1 < 0x1.fp-31 */
    s = (s - 1)>>6;
    /* s < sqrt(m) < s + 0x1.08p-23 */

    /* compute nearest rounded result.  */
    uint32_t d0, d1;
    float y;
    d0 = (m << 16) - s*s;
    d1 = s - d0;
    s += d1 >> 31;
    s &= 0x007fffff;
    s |= ey;
    y = asfloat(s);
    return y;
}
```

> `ix -= 23U << 23` 是 musl 原版的补偿：它故意让指数位算术回绕，后续 `ey = ix >> 1` 的算术把回绕抵消掉。改成整数重建时**必须保留这一行**，否则次正规数输入的指数全错。

> `sqrt` / `sqrtf` 的特殊值分支顺序必须与 spec 的 errno 表一致：**先判 NaN（任意符号）返回固定 quiet NaN 且不动 `errno`，再判负数（含 `-inf`）置 `EDOM`**。用符号位直接判负会把 `-NaN` 误判成域错误。

```c
 *
 * A standalone host program with NO OS01 include paths: it can only use
 * the host's <math.h> and the host's libm.  It evaluates the shared
 * case table and prints one row per case:
 *
 *     <fn> <width> <result-bits-hex> <errno>
 *
 * errno is set to a sentinel before every call so the reference's
 * errno discipline is visible too.
 */
#include <errno.h>
#include <math.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

#include "../data/libm_cases.h"

#define SENTINEL 0x5a5a

static double bits_to_double(uint64_t u)
{
    double d;
    memcpy(&d, &u, sizeof(d));
    return d;
}

static float bits_to_float(uint64_t u)
{
    float f;
    uint32_t v = (uint32_t)u;
    memcpy(&f, &v, sizeof(f));
    return f;
}

static uint64_t double_to_bits(double d)
{
    uint64_t u;
    memcpy(&u, &d, sizeof(u));
    return u;
}

static uint32_t float_to_bits(float f)
{
    uint32_t v;
    memcpy(&v, &f, sizeof(v));
    return v;
}

int main(void)
{
    for (int i = 0; i < LIBM_CASE_COUNT; i++) {
        const struct libm_case *c = &libm_cases[i];
        uint64_t result = 0;

        errno = SENTINEL;
        if (c->width == 64) {
            double x = bits_to_double(c->a);
            double y = bits_to_double(c->b);
            double r;
            if (!strcmp(c->fn, "fabs"))            r = fabs(x);
            else if (!strcmp(c->fn, "copysign"))  r = copysign(x, y);
            else if (!strcmp(c->fn, "floor"))     r = floor(x);
            else if (!strcmp(c->fn, "ceil"))      r = ceil(x);
            else if (!strcmp(c->fn, "trunc"))     r = trunc(x);
            else if (!strcmp(c->fn, "round"))     r = round(x);
            else if (!strcmp(c->fn, "fmod"))      r = fmod(x, y);
            else if (!strcmp(c->fn, "sqrt"))      r = sqrt(x);
            else { fprintf(stderr, "unknown fn %s\n", c->fn); return 2; }
            result = double_to_bits(r);
        } else {
            float x = bits_to_float(c->a);
            float y = bits_to_float(c->b);
            float r;
            if (!strcmp(c->fn, "fabsf"))           r = fabsf(x);
            else if (!strcmp(c->fn, "copysignf"))  r = copysignf(x, y);
            else if (!strcmp(c->fn, "floorf"))     r = floorf(x);
            else if (!strcmp(c->fn, "ceilf"))      r = ceilf(x);
            else if (!strcmp(c->fn, "truncf"))     r = truncf(x);
            else if (!strcmp(c->fn, "roundf"))     r = roundf(x);
            else if (!strcmp(c->fn, "fmodf"))      r = fmodf(x, y);
            else if (!strcmp(c->fn, "sqrtf"))      r = sqrtf(x);
            else { fprintf(stderr, "unknown fn %s\n", c->fn); return 2; }
            result = float_to_bits(r);
        }
        printf("%s %d %016llx %d\n", c->fn, c->width,
               (unsigned long long)result, errno);
    }
    return 0;
```

```c
/* the expected result of a kind-2 row: the input NaN survives and only
 * the sign bit may move */
static uint64_t expect_kind2(const struct libm_case *c)
{
    uint64_t sign = (c->width == 64) ? (1ULL << 63) : (1U << 31);
    uint64_t magmask = (c->width == 64) ? 0x7fffffffffffffffULL : 0x7fffffffU;
    uint64_t a_mag = c->a & magmask;

    if (!strcmp(c->fn, "fabs") || !strcmp(c->fn, "fabsf"))
        return a_mag;                        /* the sign bit is cleared */
    if (!strcmp(c->fn, "copysign") || !strcmp(c->fn, "copysignf"))
        return a_mag | (c->b & sign);       /* magnitude of a, sign of b */
    return c->a;                            /* the rounding functions
                                             * return a NaN unchanged */
}

TEST_FUNC(test_libm_matches_glibc_oracle) {
    /* run the oracle and collect its rows */
    int fds[2];
    assert_eq(0, pipe(fds));
    int64_t pid = fork();
    if (pid == 0) {
        close(fds[0]);
        dup2(fds[1], 1);
        dup2(fds[1], 2);
        close(fds[1]);
        char *argv[2] = { (char *)LIBM_ORACLE_PATH, NULL };
        execv(LIBM_ORACLE_PATH, argv);
        _exit(127);
    }
    close(fds[1]);
    static char out[64 * 1024];
    size_t total = 0;
    for (;;) {
        if (total + 1 >= sizeof(out)) break;
        ssize_t n = read(fds[0], out + total, sizeof(out) - 1 - total);
        if (n <= 0) break;
        total += (size_t)n;
    }
    close(fds[0]);
    out[total] = 0;
    assert_true(total > 0);

    /* parse "<fn> <width> <bits> <errno>" rows */
    char *line = out;
    int row = 0;
    while (line && *line && row < LIBM_CASE_COUNT) {
        char *nl = strchr(line, '\n');
        if (nl) *nl = 0;
        char fn[32];
        int width = 0;
        unsigned long long bits = 0;
        int ref_errno = 0;
        if (sscanf(line, "%31s %d %llx %d", fn, &width, &bits, &ref_errno) != 4) {
            FAIL("libm oracle row", line);
            break;
        }
        const struct libm_case *c = &libm_cases[row];
        assert_str_eq(fn, c->fn);

        os01_errno = 0x5a5a;
        uint64_t got = call_os01(c);

        if (c->kind == 0 && (uint64_t)bits != got) {
            printf("  [libm] %s mismatch: os01=%016llx glibc=%016llx "
                   "os01_errno=%d glibc_errno=%d\n",
                   c->fn, (unsigned long long)got, (unsigned long long)bits,
                   os01_errno, ref_errno);
        }

        if (c->kind == 0) {
            /* a finite/infinite/zero result: bit-identical to glibc */
            assert_eq((uint64_t)bits, got);
            assert_eq(0x5a5a, os01_errno);      /* no errno on the happy path */
        } else if (c->kind == 1) {
            /* the fixed quiet NaN */
            uint64_t want = (c->width == 64) ? OS01_QNAN_D : OS01_QNAN_F;
            assert_eq(want, got);
            if (!strcmp(c->fn, "sqrt") || !strcmp(c->fn, "sqrtf") ||
                !strcmp(c->fn, "fmod") || !strcmp(c->fn, "fmodf")) {
                /* a domain error sets EDOM, a NaN operand does not */
                int expect_edom = strstr(c->fn, "sqrt")
                                  ? (c->a >> 63)          /* negative: domain */
                                  : 1;                    /* fmod: domain */
                if (!strcmp(c->fn, "fmod") || !strcmp(c->fn, "fmodf")) {
                    uint64_t aexp = (c->a >> 52) & 0x7ff;
                    uint64_t bexp = (c->b >> 52) & 0x7ff;
                    int a_nan = (aexp == 0x7ff) && (c->a & 0xfffffffffffffULL);
                    int b_nan = (bexp == 0x7ff) && (c->b & 0xfffffffffffffULL);
                    expect_edom = !a_nan && !b_nan;
                }
                assert_eq(expect_edom ? EDOM : 0x5a5a, os01_errno);
            }
        } else {
            /* kind 2: the input NaN payload survives, only the sign moves */
            assert_eq(expect_kind2(c), got);
        }
        row++;
        line = nl ? nl + 1 : NULL;
    }
    assert_eq(LIBM_CASE_COUNT, row);
}
```

```c
 *
 * Every one of the sixteen symbols is called and its result is checked
 * against a hard-coded bit pattern: the guest has no reference to
 * compare against, so the expectations live here.

#include "systest.h"

#include <math.h>
#include <stdint.h>
#include <string.h>

#define QNAN_D 0x7ff8000000000000ULL
#define QNAN_F 0x7fc00000U

static uint64_t dbits(double d)
{
    uint64_t u;
    memcpy(&u, &d, sizeof(u));
    return u;
}

static uint32_t fbits(float f)
{
    uint32_t v;
    memcpy(&v, &f, sizeof(v));
    return v;
}

static double fromd(uint64_t u)
{
    double d;
    memcpy(&d, &u, sizeof(d));
    return d;
}

static float fromf(uint32_t u)
{
    float f;
    memcpy(&f, &u, sizeof(f));
    return f;
}

void test_libm(void)
{
    /* ── fabs / fabsf: the sign bit only ───────────────────────── */
    CHECK3(dbits(fabs(-0.0)) == 0x0000000000000000ULL, "libm fabs", "-0.0 -> +0.0");
    CHECK3(dbits(fabs(-1.5)) == 0x3ff8000000000000ULL, "libm fabs", "-1.5 -> 1.5");
    CHECK3(dbits(fabs(-INFINITY)) == 0x7ff0000000000000ULL, "libm fabs", "-inf -> +inf");
    CHECK3(dbits(fabs(fromd(0x7ff8000000001234ULL))) == 0x7ff8000000001234ULL,
           "libm fabs", "NaN payload survives");
    CHECK3(fbits(fabsf(-0.0f)) == 0x00000000U, "libm fabsf", "-0.0f -> +0.0f");

    /* ── copysign / copysignf: magnitude from a, sign from b ────── */
    CHECK3(dbits(copysign(1.5, -2.0)) == 0xbff8000000000000ULL,
           "libm copysign", "sign taken from y");
    CHECK3(dbits(copysign(fromd(0x7ff8000000001234ULL), -1.0)) == 0xfff8000000001234ULL,
           "libm copysign", "NaN payload survives");

    /* ── floor / ceil / trunc / round ──────────────────────────── */
    CHECK3(dbits(floor(1.5)) == 0x3ff0000000000000ULL, "libm floor", "1.5 -> 1");
    CHECK3(dbits(floor(-1.5)) == 0xbff0000000000000ULL, "libm floor", "-1.5 -> -2");
    CHECK3(dbits(floor(-0.0)) == 0x8000000000000000ULL, "libm floor", "-0.0 keeps its sign");
    CHECK3(dbits(ceil(1.5)) == 0x4000000000000000ULL, "libm ceil", "1.5 -> 2");
    CHECK3(dbits(ceil(-1.5)) == 0xbff0000000000000ULL, "libm ceil", "-1.5 -> -1");
    CHECK3(dbits(trunc(-1.5)) == 0xbff0000000000000ULL, "libm trunc", "-1.5 -> -1");
    CHECK3(dbits(round(1.5)) == 0x4000000000000000ULL, "libm round", "1.5 -> 2 (half away)");
    CHECK3(dbits(round(-1.5)) == 0xc000000000000000ULL, "libm round", "-1.5 -> -2 (half away)");
    CHECK3(dbits(round(2.5)) == 0x4004000000000000ULL, "libm round", "2.5 -> 3 (half away)");
    CHECK3(dbits(round(-0.4)) == 0x8000000000000000ULL, "libm round", "-0.4 -> -0.0");

    /* ── fmod: the exact remainder ─────────────────────────────── */
    CHECK3(dbits(fmod(5.5, 2.0)) == 0x3ff6000000000000ULL, "libm fmod", "5.5 % 2 = 1.5");
    CHECK3(dbits(fmod(-5.5, 2.0)) == 0xbff6000000000000ULL, "libm fmod", "-5.5 % 2 = -1.5");

    /* ── sqrt: correctly rounded ───────────────────────────────── */
    CHECK3(dbits(sqrt(4.0)) == 0x4010000000000000ULL, "libm sqrt", "sqrt(4) = 2");
    CHECK3(dbits(sqrt(0.25)) == 0x3fd0000000000000ULL, "libm sqrt", "sqrt(0.25) = 0.5");

    /* ── the errno contract ────────────────────────────────────── */
    errno = 0;
    (void)sqrt(-1.0);
    CHECK3(errno == EDOM, "libm errno", "sqrt(-1) sets EDOM");
    errno = 0;
    (void)fmod(5.0, 0.0);
    CHECK3(errno == EDOM, "libm errno", "fmod(x, 0) sets EDOM");
    errno = 0;
    (void)fmod(INFINITY, 2.0);
    CHECK3(errno == EDOM, "libm errno", "fmod(inf, x) sets EDOM");
    errno = 0;
    (void)sqrt(NAN);
    CHECK3(errno == 0, "libm errno", "sqrt(NaN) leaves errno alone");
    errno = 0;
    (void)floor(1.5);
    CHECK3(errno == 0, "libm errno", "floor leaves errno alone");

    /* ── the fixed quiet NaN ───────────────────────────────────── */
    errno = 0;
    CHECK3(dbits(sqrt(-1.0)) == QNAN_D, "libm nan", "sqrt(-1) -> fixed quiet NaN");
    errno = 0;
    CHECK3(dbits(fmod(1.0, 0.0)) == QNAN_D, "libm nan", "fmod(x, 0) -> fixed quiet NaN");
    errno = 0;
    CHECK3(fbits(sqrtf(-1.0f)) == QNAN_F, "libm nan", "sqrtf(-1) -> fixed quiet NaN");
}
```

- [ ] **Step 3: 写反汇编审计 `scripts/audit_libm_asm.py`**

```python
#!/usr/bin/env python3
"""audit_libm_asm.py — the libm objects must not execute FP arithmetic.

The vendored musl subset computes every result with integer bit
operations.  The audit disassembles the objects and rejects the
floating-point arithmetic, compare, round and convert instructions;
parameter and result moves through the FP registers are allowed (the
userland ABI passes doubles in xmm0 / v0).

Usage:
    python3 scripts/audit_libm_asm.py --objects <dir> [--profile x86_64-clang]
"""
from __future__ import annotations

import argparse
import os
import re
import subprocess
import sys

BANNED = [
    # x86_64 SSE / AVX arithmetic, compare, round, convert
    r"\b(add|sub|mul|div|sqrt|max|min|cmp|comi|ucomi|round|cvt)(p?s[sd]|ss|pd)\b",
    r"\bv(add|sub|mul|div|sqrt|max|min|cmp|round|cvt)\w*\b",
    # aarch64 scalar/vector FP arithmetic
    r"\bf(add|sub|mul|div|sqrt|max|min|cmp|cmpe|cmpz|neg|abs|cvt|fcvt|scvtf|ucvtf|frint[a-z]*)\w*\b",
    r"\b(fcmeq|fcsel|fccmp|fccmpe|fcmla|fcmlt|fcmlt)\b",
    # x87
    r"\bf(ld|st|add|sub|mul|div|sqrt|abs|chs|prem|patan)\w*\b",
]

MNEMONIC = re.compile(r"^\s+[0-9a-f]+:\s+(?:[0-9a-f]{2} )+\s*\t([a-z0-9.]+)")


def disassemble(path: str) -> str:
    tool = os.environ.get("LLVM_OBJDUMP", "llvm-objdump")
    try:
        return subprocess.run([tool, "-d", "--no-show-raw-insn", path],
                              check=True, capture_output=True, text=True).stdout
    except (subprocess.CalledProcessError, FileNotFoundError) as exc:
        print(f"[asm-audit] ERROR: cannot disassemble {path}: {exc}", file=sys.stderr)
        raise SystemExit(2)


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--objects", required=True,
                    help="the directory holding the compiled libm objects")
    args = ap.parse_args()

    objects = sorted(
        os.path.join(args.objects, name)
        for name in os.listdir(args.objects)
        if name.endswith(".o")
    )
    if not objects:
        print(f"[asm-audit] ERROR: no objects under {args.objects}", file=sys.stderr)
        return 1

    errors: list[str] = []
    for path in objects:
        text = disassemble(path)
        for line in text.splitlines():
            match = MNEMONIC.match(line)
            if not match:
                continue
            mnemonic = match.group(1)
            for pattern in BANNED:
                if re.search(pattern, mnemonic):
                    errors.append(f"{os.path.basename(path)}: {mnemonic}  ({line.strip()})")
                    break

    for err in errors:
        print(f"[asm-audit] ERROR: floating-point instruction in libm: {err}")
    if errors:
        print(f"[asm-audit] {len(errors)} banned instruction(s) found")
        return 1
    print(f"[asm-audit] {len(objects)} libm object(s): no floating-point arithmetic")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
